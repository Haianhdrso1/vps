local function GetNearestTitan()
    local nearest = nil
    local minDist = Settings.FarmRange
    for _, v in pairs(workspace:GetDescendants()) do
        if v:IsA("Model") and v.Name:find("Titan") then
            local root = v:FindFirstChild("HumanoidRootPart")
            if root and v:FindFirstChild("Humanoid") and v.Humanoid.Health > 0 then
                local dist = (root.Position - Character.HumanoidRootPart.Position).Magnitude
                if dist < minDist then
                    minDist = dist
                    nearest = v
                end
            end
        end
    end
    return nearest
end

local function GetTitansInRange(range)
    local list = {}
    for _, v in pairs(workspace:GetDescendants()) do
        if v:IsA("Model") and v.Name:find("Titan") then
            local root = v:FindFirstChild("HumanoidRootPart")
            if root and v:FindFirstChild("Humanoid") and v.Humanoid.Health > 0 then
                local dist = (root.Position - Character.HumanoidRootPart.Position).Magnitude
                if dist < range then
                    table.insert(list, v)
                end
            end
        end
    end
    return list
end

local function AttackTarget(target)
    if not target then return end
    local root = target:FindFirstChild("HumanoidRootPart")
    if not root then return end
    -- Di chuyển đến gần
    Character.Humanoid:MoveTo(root.Position + Vector3.new(0, 0, 5))
    -- Tấn công
    VirtualInputManager:SendMouseButtonEvent(Enum.UserInputType.MouseButton1, 0, 0, true, game, 0)
    task.wait(Settings.AttackSpeed)
    VirtualInputManager:SendMouseButtonEvent(Enum.UserInputType.MouseButton1, 0, 0, false, game, 0)
end

local function TeleportTo(pos)
    if Character:FindFirstChild("HumanoidRootPart") then
        Character.HumanoidRootPart.CFrame = CFrame.new(pos)
    end
end

-- 1. Auto Farm
local function AutoFarm()
    if not Settings.AutoFarm then return end
    local titan = GetNearestTitan()
    if titan then
        AttackTarget(titan)
    else
        Character.Humanoid:MoveTo(Vector3.new(math.random(-300,300), 0, math.random(-300,300)))
    end
end

-- 2. Kill Aura
local function KillAura()
    if not Settings.KillAura then return end
    local titans = GetTitansInRange(Settings.FarmRange)
    for _, titan in pairs(titans) do
        AttackTarget(titan)
        task.wait(0.05)
    end
end

-- 3. Auto Raid

        task.wait(1)
    end
end

local espObjects = {}
local function TitanESP()
    if not Settings.TitanESP then
        for _, v in pairs(workspace:GetDescendants()) do
            if v:IsA("Highlight") and v.Name == "ESP_Titan" then
                v:Destroy()
            end
        end
        return
    end
    for _, v in pairs(workspace:GetDescendants()) do
        if v:IsA("Model") and v.Name:find("Titan") then
            local root = v:FindFirstChild("HumanoidRootPart")
            if root and not v:FindFirstChild("ESP_Titan") then
                local highlight = Instance.new("Highlight")

                highlight.Parent = v

                highlight.FillColor = Color3.fromRGB(255, 0, 0)
                highlight.FillTransparency = 0.25
                highlight.OutlineColor = Color3.fromRGB(255, 255, 255)
                highlight.OutlineTransparency = 0
                highlight.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
            end
        end
    end
end

-- 8. Auto Serum
local function AutoSerum()
    if not Settings.AutoSerum then return end
    local backpack = LocalPlayer:FindFirstChild("Backpack")
    if backpack then
        for _, item in pairs(backpack:GetChildren()) do
            if item:IsA("Tool") and (item.Name:find("Serum") or item.Name:find("Shard")) then
                item.Parent = Character
                task.wait(0.5)
                if item:IsA("Tool") then
                    item:Activate()

                task.wait(0.5)
                item.Parent = backpack
            end
        end
    end
end

-- 9. Auto Shift Titan (Titan Shift)
local function AutoShift()
    if not Settings.AutoShift then return end
    -- Tìm skill Titan Shift trong hotbar
    local hotbar = LocalPlayer:FindFirstChild("Hotbar")
    if hotbar then
        for _, slot in pairs(hotbar:GetChildren()) do
            if slot:IsA("Frame") and slot:FindFirstChild("Icon") and slot.Icon:FindFirstChild("ImageLabel") then
                local img = slot.Icon.ImageLabel
                if img and img.Image and img.Image:find("Titan") then
                    slot:Invoke()
                    task.wait(1)
                    break
                end
            end
        end
    end
    -- Cách khác: tìm trong backpack
    local backpack = LocalPlayer:FindFirstChild("Backpack")
    if backpack then
        for _, item in pairs(backpack:GetChildren()) do
            if item:IsA("Tool") and (item.Name:find("Titan") or item.Name:find("Shift")) then
                item.Parent = Character
                task.wait(0.5)
                item:Activate()
                task.wait(1)
                item.Parent = backpack
                break
            end
        end
    end
end

            end
        end
    end
end
